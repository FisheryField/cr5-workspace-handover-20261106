dos2unix /tmp/tUpdate/control/toolUtils.sh
chmod +x /tmp/tUpdate/control/toolUtils.sh
source /tmp/tUpdate/control/toolUtils.sh
execute_command "pwd"

updateFile="/tmp/tUpdate/control/bin.tar.gz"

memsize=$(free | awk 'NR==2{print $2}')
memsize=$((memsize))
if [ $memsize -gt 1025000 ]; then
    echo "20480" > ${runResult_file}
    echo "not enough memory" >> ${controlRunProcess_file}
    exit 1
else 
    if [ -f ${updateFile} ]; then   
        echo "the memory is enough" >> ${controlRunProcess_file}
        if [ -f "/home/ftp/PlcLogic/Recipe/SavePos.txt"]; then
            rm -rf /home/ftp/PlcLogic/Recipe/SavePos.txt
        fi
        execute_command "mv ${control_base_path}/bin.tar.gz /tmp/"
        execute_command "mv ${control_base_path}/afterUpdate.sh /tmp/"
        execute_command "chmod +x /tmp/afterUpdate.sh"
        execute_command "dos2unix /tmp/afterUpdate.sh"
        /tmp/afterUpdate.sh >>${controlRunProcess_file} 2>&1
        if [ $? -eq 0 ]; then
            echo "run afterUpdate.sh successfully" >> ${controlRunProcess_file}
        else 
            echo "20482" > ${runResult_file}
            echo "failed to run afterUpdate.sh" >> ${controlRunProcess_file}
			exit 1
        fi
        execute_command "sync"
        exit 0
    else 
        echo "20481" > ${runResult_file}
        echo "failed to find ${updateFile}" >> ${controlRunProcess_file}
        exit 1     
    fi
fi
